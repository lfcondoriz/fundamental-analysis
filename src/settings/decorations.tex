\usepackage{tikz}

% Un símbolo decorativo (tipo "cruz" rotada)
\newcommand{\andreas}[5][1.0]{%
    \begin{scope}[shift={(#2,#3)},rotate=45,scale=#4,transparency group,opacity=#1]
        \fill[fill=#5] (0,0) -- (0,1) -- (3,1) -- (3,0);
        \fill[fill=#5] (1,-1) -- (2,-1) -- (2,2) -- (1,2);
    \end{scope}
}

%% Línea decorativa horizontal gruesa
\newcommand{\andreasline}{%
    \centering
    \begin{tikzpicture}
        \pgfmathsetmacro{\andreasstep}{0.5}
        \pgfmathsetmacro{\andreaswidth}{\linewidth / 1cm}
        \pgfmathtruncatemacro{\andreascount}{max(1, floor(\andreaswidth / \andreasstep))}
        \pgfmathsetmacro{\andreasoffset}{(\andreaswidth - (\andreascount * \andreasstep)) / 2}
        \foreach \n in {1,...,\andreascount} {
            \pgfmathsetmacro{\andreasopacity}{0.9 * sin((\n + 1) * 180 / (\andreascount + 2))}
            \andreas[\andreasopacity]{\andreasoffset + (\n - 0.5) * \andreasstep}{0}{0.10}{black}
        }
    \end{tikzpicture}
}

%% Línea decorativa más delgada (slim)
\newcommand{\andreaslineslim}{%
    \centering
    \begin{tikzpicture}[baseline]
        \pgfmathsetmacro{\andreasstep}{0.20}
        \pgfmathsetmacro{\andreaswidth}{\linewidth / 1cm}
        \pgfmathtruncatemacro{\andreascount}{max(1, floor(\andreaswidth / \andreasstep))}
        \pgfmathsetmacro{\andreasoffset}{(\andreaswidth - (\andreascount * \andreasstep)) / 2}
        \foreach \n in {1,...,\andreascount} {
            \andreas[0.75]{\andreasoffset + (\n - 0.5) * \andreasstep}{0}{0.04}{black}
        }
    \end{tikzpicture}
}
