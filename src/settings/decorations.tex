\usepackage{tikz}

% Símbolo decorativo (cruz rotada 45°)
\newcommand{\andreas}[5][1.0]{%
    \path[fill=#5,shift={(#2,#3)},rotate=45,scale=#4,transparency group,opacity=#1] (0,0) rectangle (3,1);
    \path[fill=#5,shift={(#2,#3)},rotate=45,scale=#4,transparency group,opacity=#1] (1,-1) rectangle (2,2);
}

%% Línea decorativa horizontal
\newcommand{\andreasline}{%
    \par\medskip
    \centering
    \begin{tikzpicture}
        \pgfmathsetmacro{\andreasstep}{0.5}
        \pgfmathsetmacro{\andreaswidth}{\linewidth / 1cm}
        \pgfmathtruncatemacro{\andreascount}{max(1, floor(\andreaswidth / \andreasstep))}
        \pgfmathsetmacro{\andreasoffset}{(\andreaswidth - (\andreascount * \andreasstep)) / 2}
        \foreach \n in {1,...,\andreascount} {
            \pgfmathsetmacro{\andreasopacity}{0.9 * sin((\n + 1) * 180 / (\andreascount + 2))}
            \pgfmathsetmacro{\andreasx}{\andreasoffset + (\n - 0.5) * \andreasstep}
            \andreas[\andreasopacity]{\andreasx}{0}{0.10}{black}
        }
    \end{tikzpicture}
    \par\medskip
}

%% Línea decorativa más delgada (slim)
\newcommand{\andreaslineslim}{%
    \par\medskip
    \centering
    \begin{tikzpicture}[baseline]
        \pgfmathsetmacro{\andreasstep}{0.20}
        \pgfmathsetmacro{\andreaswidth}{\linewidth / 1cm}
        \pgfmathtruncatemacro{\andreascount}{max(1, floor(\andreaswidth / \andreasstep))}
        \pgfmathsetmacro{\andreasoffset}{(\andreaswidth - (\andreascount * \andreasstep)) / 2}
        \foreach \n in {1,...,\andreascount} {
            \pgfmathsetmacro{\andreasx}{\andreasoffset + (\n - 0.5) * \andreasstep}
            \andreas[0.75]{\andreasx}{0}{0.04}{black}
        }
    \end{tikzpicture}
    \par\medskip
}
